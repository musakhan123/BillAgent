# BillAgent – Full Requirements
# This is a Node.js project. To install npm packages run: npm install
# This file documents ALL requirements needed to run on a new system.

# ─────────────────────────────────────────────
# 1. RUNTIME
# ─────────────────────────────────────────────
Node.js >= 18.0.0   # LTS recommended; required for native fetch, --watch flag
npm >= 9.0.0        # bundled with Node.js 18+

# ─────────────────────────────────────────────
# 2. SYSTEM-LEVEL DEPENDENCIES (install manually)
# ─────────────────────────────────────────────

# Poppler – required by pdf-poppler to convert PDF pages to images
#   Windows : https://github.com/oschwartz10612/poppler-windows/releases
#             Extract and add the bin/ folder to PATH
#   macOS   : brew install poppler
#   Ubuntu  : sudo apt install poppler-utils

# ─────────────────────────────────────────────
# 3. NPM PACKAGES  (installed via: npm install)
# ─────────────────────────────────────────────
@langchain/core@^1.1.45            # LangChain core
@langchain/google-genai@^2.1.30    # LangChain × Gemini integration
dotenv@^16.4.5                     # .env loader
express@^4.19.2                    # Web framework
multer@^1.4.5-lts.1                # File upload middleware
pdf-poppler@^0.2.3                 # PDF → image (wraps Poppler CLI)
puppeteer-core@^24.43.0            # Headless browser automation
